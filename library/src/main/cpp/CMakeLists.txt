cmake_minimum_required(VERSION 3.22.1)
project(weapp_container_android LANGUAGES CXX)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(SDK_TARGET aarch64-linux-android)
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(SDK_TARGET x86_64-linux-android)
else()
  message(FATAL_ERROR "Unsupported ABI: ${ANDROID_ABI}")
endif()
set(SDK_DIR "${WEAPP_SDK_NATIVE_DIR}/${SDK_TARGET}")
foreach(FILE libweapp_container_sdk.so weapp_container_sdk.h)
  if(NOT EXISTS "${SDK_DIR}/${FILE}")
    message(FATAL_ERROR "Missing SDK artifact: ${SDK_DIR}/${FILE}")
  endif()
endforeach()
add_library(weapp_container_sdk SHARED IMPORTED)
set_target_properties(weapp_container_sdk PROPERTIES IMPORTED_LOCATION "${SDK_DIR}/libweapp_container_sdk.so" IMPORTED_SONAME "libweapp_container_sdk.so")
add_library(weapp_container_android SHARED weapp_jni.cpp)
target_include_directories(weapp_container_android PRIVATE "${SDK_DIR}")
target_link_libraries(weapp_container_android PRIVATE weapp_container_sdk)
target_link_options(weapp_container_android PRIVATE "-Wl,-z,max-page-size=16384")
